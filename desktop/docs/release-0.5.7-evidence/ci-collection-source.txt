#!/usr/bin/env python3
"""Collect immutable final 0.5.7 GitHub Actions evidence; no test or publish actions."""
import hashlib
import json
import subprocess
from pathlib import Path
ROOT = Path('/workspace/pureTV/desktop/docs/release-0.5.7-evidence')
RUN = 43696998459
SOURCE = '0f5c4ffa535b6f5f6c49758ca5be80290bfec21e'
TAG = '3b26212b53022a54dffa3f0177f9d99373492b35'
REPO = 'ecojust/pureTV'
def gh(*args):
    return subprocess.check_output(['gh', *args], cwd='/workspace/pureTV')
def save(name, value):
    (ROOT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
run = json.loads(gh('run', 'view', str(RUN), '--repo', REPO, '--json', 'databaseId,status,conclusion,jobs,headSha,url,createdAt,updatedAt,headBranch,event,name,workflowName'))
assert run['status'] == 'completed' and run['headSha'] == SOURCE
save('ci-run-final.json', run)
metadata = json.loads(gh('api', f'repos/{REPO}/actions/runs/{RUN}'))
assert metadata['head_sha'] == SOURCE and metadata['id'] == RUN
save('ci-run-metadata.json', metadata)
ref = json.loads(gh('api', f'repos/{REPO}/git/ref/tags/desktop-v0.5.7'))
assert ref['object']['sha'] == TAG
save('ci-tag-ref.json', ref)
tag = json.loads(gh('api', f'repos/{REPO}/git/tags/{TAG}'))
assert tag['sha'] == TAG and tag['object']['sha'] == SOURCE
save('ci-tag-metadata.json', tag)
index = {}
for job in run['jobs']:
    name = job['name']
    process = subprocess.run(['gh', 'run', 'view', str(RUN), '--repo', REPO, '--job', str(job['databaseId']), '--log'], cwd='/workspace/pureTV', capture_output=True)
    if process.returncode:
        index[name] = {'jobId':job['databaseId'], 'logRetrieved':False, 'error':process.stderr.decode(errors='replace').strip()}
        save('ci-raw-log-index.json', index)
        print(name, 'LOG UNAVAILABLE', index[name]['error'], flush=True)
        continue
    raw = process.stdout
    stem = 'publish' if name == 'Publish desktop release' else name
    path = ROOT / ('ci-' + stem + '.txt')
    path.write_bytes(raw)
    index[name] = {'jobId':job['databaseId'], 'logRetrieved':True, 'file':path.name, 'logBytes':len(raw), 'logSha256':hashlib.sha256(raw).hexdigest()}
    save('ci-raw-log-index.json', index)
    print(name, job['conclusion'], len(raw), flush=True)
save('ci-raw-log-index.json', index)
