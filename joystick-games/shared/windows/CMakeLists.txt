cmake_minimum_required(VERSION 3.20)
project(PureTVGame LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PURETV_DEPS)
  message(FATAL_ERROR "PURETV_DEPS must point to the hash-verified dependency directory")
endif()
add_executable(protocol-test tests/protocol.test.cpp)
target_include_directories(protocol-test PRIVATE "${PURETV_DEPS}" Sources)
add_executable(preferences-test tests/preferences.test.cpp)
target_include_directories(preferences-test PRIVATE "${PURETV_DEPS}" Sources)
add_executable(entry-url-test tests/entry-url.test.cpp)
target_include_directories(entry-url-test PRIVATE Sources)
add_executable(diagnostic-log-test tests/diagnostic-log.test.cpp)
enable_testing()
add_test(NAME protocol COMMAND protocol-test)
add_test(NAME preferences COMMAND preferences-test)
add_test(NAME entry-url COMMAND entry-url-test)
add_test(NAME diagnostic-log COMMAND diagnostic-log-test)
if(WIN32)
  add_executable(PureTVGame WIN32 Sources/main.cpp Sources/app.manifest)
  set_property(TARGET PureTVGame protocol-test preferences-test entry-url-test diagnostic-log-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(PureTVGame PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000007)
  target_compile_options(PureTVGame PRIVATE /W4 /utf-8 /EHsc)
  target_include_directories(PureTVGame PRIVATE "${PURETV_DEPS}" "${PURETV_DEPS}/webview2/build/native/include")
  target_link_libraries(PureTVGame PRIVATE "${PURETV_DEPS}/webview2/build/native/x64/WebView2LoaderStatic.lib" ws2_32 dnsapi iphlpapi ole32 oleaut32 shlwapi shell32 user32 advapi32 version)
  target_link_options(PureTVGame PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/Sources/app.manifest")
endif()
